cmake_minimum_required(VERSION 3.20)
project(LaserSpoutReceiver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Spout2 receiver requires Windows")
endif()
if(NOT DEFINED SPOUT_SDK_ROOT)
  message(FATAL_ERROR "Pass -DSPOUT_SDK_ROOT=C:/path/to/Spout2")
endif()
add_executable(LaserSpoutReceiver spout-receiver.cpp)
target_include_directories(LaserSpoutReceiver PRIVATE "${SPOUT_SDK_ROOT}/SPOUTSDK/SpoutLibrary")
set_target_properties(LaserSpoutReceiver PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/bin")
